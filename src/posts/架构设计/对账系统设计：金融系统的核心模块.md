# 对账系统设计：金融系统的核心模块

> 对账是金融系统的核心模块，是保障资金安全、数据一致性的最后一道防线。
> 本文从金融业务视角出发，系统介绍对账的完整体系，包括对账分类、架构设计、核心算法、账务处理和生产实践，帮助你构建企业级对账系统。

## 一、对账概述

### 1.1 什么是对账

```
对账定义：

对账(Reconciliation)是指将两个或多个数据源的数据进行核对，发现差异并分析原因的过程。

在金融系统中，对账的核心目标：
├─ 资金安全：确保资金流水准确无误
├─ 数据一致：保证各系统间数据一致
├─ 风险控制：及时发现异常交易
└─ 合规要求：满足监管审计要求
```

### 1.2 对账分类体系

| 分类维度 | 类型 | 说明 | 典型场景 |
|----------|------|------|----------|
| **按范围** | 内部对账 | 系统内部数据核对 | 账户余额、总分核对 |
| | 外部对账 | 与外部机构数据核对 | 银行流水、渠道对账 |
| | 跨系统对账 | 多系统间数据核对 | 支付系统与账务系统 |
| **按频率** | 实时对账 | 交易发生时实时核对 | 支付结果确认 |
| | 准实时对账 | 分钟级/小时级核对 | 风险监控 |
| | 批量对账 | T+1或定时批量核对 | 日终对账、月度对账 |
| **按业务** | 交易对账 | 交易流水核对 | 支付、转账交易 |
| | 账务对账 | 账户余额核对 | 总分核对、内部往来 |
| | 资金对账 | 资金流水核对 | 银行资金、备付金 |
| | 渠道对账 | 渠道数据核对 | 第三方支付渠道 |

```java
/**
 * 对账数据采集规范
 */
public interface ReconciliationDataSource {
    
    /**
     * 获取对账数据
     * @param date 对账日期
     * @param dataType 数据类型
     * @return 对账数据列表
     */
    List<ReconciliationData> fetchData(LocalDate date, DataType dataType);
    
    /**
     * 数据格式转换
     */
    default List<ReconciliationData> convert(List<RawData> rawData) {
        return rawData.stream()
            .map(this::convertSingle)
            .collect(Collectors.toList());
    }
    
    /**
     * 数据校验
     */
    default ValidationResult validate(List<ReconciliationData> data) {
        ValidationResult result = new ValidationResult();
        
        // 完整性校验
        if (data == null || data.isEmpty()) {
            result.addError("数据为空");
            return result;
        }
        
        // 字段完整性校验
        for (ReconciliationData item : data) {
            if (item.getTransactionId() == null) {
                result.addError("交易流水号为空: " + item);
            }
            if (item.getAmount() == null || item.getAmount().compareTo(BigDecimal.ZERO) <= 0) {
                result.addError("金额异常: " + item);
            }
        }
        
        // 唯一性校验
        long uniqueCount = data.stream()
            .map(ReconciliationData::getTransactionId)
            .distinct()
            .count();
        if (uniqueCount != data.size()) {
            result.addError("存在重复数据");
        }
        
        return result;
    }
}
```

```java
@Component
public class ReconciliationRuleEngine {
    
    @Autowired
    private ReconciliationRuleRepository ruleRepository;
    
    /**
     * 获取匹配规则
     */
    public MatchRule getMatchRule(String reconciliationType) {
        List<ReconciliationRule> rules = ruleRepository
            .findByRuleTypeAndReconciliationType("MATCH", reconciliationType);
        
        MatchRule matchRule = new MatchRule();
        for (ReconciliationRule rule : rules) {
            // 解析匹配字段配置
            MatchFieldConfig config = parseMatchFields(rule.getMatchFields());
            matchRule.addConfig(config);
        }
        
        return matchRule;
    }
    
    /**
     * 获取差异检测规则
     */
    public DiffRule getDiffRule(String reconciliationType) {
        List<ReconciliationRule> rules = ruleRepository
            .findByRuleTypeAndReconciliationType("DIFF", reconciliationType);
        
        DiffRule diffRule = new DiffRule();
        for (ReconciliationRule rule : rules) {
            DiffThreshold threshold = new DiffThreshold();
            threshold.setAmountThreshold(rule.getDiffThreshold());
            diffRule.addThreshold(threshold);
        }
        
        return diffRule;
    }
    
    /**
     * 获取处理策略
     */
    public HandleStrategy getHandleStrategy(String reconciliationType, String diffType) {
        List<ReconciliationRule> rules = ruleRepository
            .findByRuleTypeAndReconciliationType("HANDLE", reconciliationType);
        
        return rules.stream()
            .filter(r -> r.getHandleStrategy().contains(diffType))
            .findFirst()
            .map(r -> parseHandleStrategy(r.getHandleStrategy()))
            .orElse(DEFAULT_HANDLE_STRATEGY);
    }
}
```

## 四、核心算法实现

### 4.1 多维度匹配算法

```java
/**
 * 多维度对账匹配算法
 */
@Component
public class MultiDimensionMatcher {
    
    /**
     * 精确匹配（基于主键）
     */
    public List<MatchResult> exactMatch(List<ReconciliationData> internal,
                                       List<ReconciliationData> external) {
        // 构建索引
        Map<String, ReconciliationData> externalMap = external.stream()
            .collect(Collectors.toMap(
                ReconciliationData::getTransactionId,
                Function.identity(),
                (v1, v2) -> v1  // 处理重复数据
            ));
        
        List<MatchResult> results = new ArrayList<>();
        Set<String> matchedExternalIds = new HashSet<>();
        
        for (ReconciliationData internalItem : internal) {
            ReconciliationData externalItem = externalMap.get(internalItem.getTransactionId());
            
            if (externalItem != null) {
                // 主键匹配成功，校验金额
                if (internalItem.getAmount().compareTo(externalItem.getAmount()) == 0) {
                    results.add(MatchResult.exactMatched(internalItem, externalItem));
                } else {
                    results.add(MatchResult.amountDiff(internalItem, externalItem));
                }
                matchedExternalIds.add(internalItem.getTransactionId());
            } else {
                results.add(MatchResult.missingExternal(internalItem));
            }
        }
        
        // 内部有但外部没有的
        for (ReconciliationData externalItem : external) {
            if (!matchedExternalIds.contains(externalItem.getTransactionId())) {
                results.add(MatchResult.missingInternal(externalItem));
            }
        }
        
        return results;
    }
    
    /**
     * 模糊匹配（基于多字段）
     */
    public List<MatchResult> fuzzyMatch(List<ReconciliationData> internal,
                                       List<ReconciliationData> external,
                                       FuzzyMatchConfig config) {
        List<MatchResult> results = new ArrayList<>();
        Set<Integer> matchedExternalIndices = new HashSet<>();
        
        for (ReconciliationData internalItem : internal) {
            Optional<MatchCandidate> bestMatch = external.stream()
                .filter((ext, idx) -> !matchedExternalIndices.contains(idx))
                .map((ext, idx) -> calculateMatchScore(internalItem, ext, config))
                .filter(candidate -> candidate.getScore() >= config.getMinScore())
                .max(Comparator.comparing(MatchCandidate::getScore));
            
            if (bestMatch.isPresent()) {
                MatchCandidate candidate = bestMatch.get();
                matchedExternalIndices.add(candidate.getExternalIndex());
                results.add(MatchResult.fuzzyMatched(internalItem, candidate.getExternal(), candidate.getScore()));
            } else {
                results.add(MatchResult.missingExternal(internalItem));
            }
        }
        
        // 外部有但内部没有的
        for (int i = 0; i < external.size(); i++) {
            if (!matchedExternalIndices.contains(i)) {
                results.add(MatchResult.missingInternal(external.get(i)));
            }
        }
        
        return results;
    }
    
    /**
     * 计算匹配得分
     */
    private MatchCandidate calculateMatchScore(ReconciliationData internal,
                                              ReconciliationData external,
                                              FuzzyMatchConfig config) {
        double score = 0.0;
        
        // 金额匹配（权重最高）
        if (internal.getAmount().compareTo(external.getAmount()) == 0) {
            score += config.getAmountWeight();
        } else {
            // 计算金额相似度
            BigDecimal diff = internal.getAmount().subtract(external.getAmount()).abs();
            BigDecimal maxAmount = internal.getAmount().max(external.getAmount());
            double similarity = 1.0 - diff.divide(maxAmount, 4, RoundingMode.HALF_UP).doubleValue();
            score += config.getAmountWeight() * similarity;
        }
        
        // 交易时间匹配
        if (isTimeMatch(internal.getTransactionTime(), external.getTransactionTime(), config.getTimeTolerance())) {
            score += config.getTimeWeight();
        }
        
        // 账户匹配
        if (Objects.equals(internal.getAccountNo(), external.getAccountNo())) {
            score += config.getAccountWeight();
        }
        
        // 交易类型匹配
        if (Objects.equals(internal.getTransactionType(), external.getTransactionType())) {
            score += config.getTypeWeight();
        }
        
        return new MatchCandidate(external, score);
    }
    
    /**
     * 批量匹配优化（大数据量场景）
     */
    public List<MatchResult> batchMatch(List<ReconciliationData> internal,
                                       List<ReconciliationData> external,
                                       int batchSize) {
        List<MatchResult> results = new ArrayList<>();
        
        // 分批处理
        List<List<ReconciliationData>> internalBatches = Lists.partition(internal, batchSize);
        List<List<ReconciliationData>> externalBatches = Lists.partition(external, batchSize);
        
        // 并行处理
        ExecutorService executor = Executors.newFixedThreadPool(Runtime.getRuntime().availableProcessors());
        List<Future<List<MatchResult>>> futures = new ArrayList<>();
        
        for (List<ReconciliationData> internalBatch : internalBatches) {
            for (List<ReconciliationData> externalBatch : externalBatches) {
                futures.add(executor.submit(() -> exactMatch(internalBatch, externalBatch)));
            }
        }
        
        // 合并结果
        for (Future<List<MatchResult>> future : futures) {
            try {
                results.addAll(future.get());
            } catch (Exception e) {
                throw new ReconciliationException("批量匹配失败", e);
            }
        }
        
        executor.shutdown();
        return results;
    }
}
```

```java
/**
 * 差异检测与分类引擎

@Component
public class DiffDetectionEngine {
    
    /**
     * 差异类型枚举
     */
    public enum DiffType {
        AMOUNT_DIFF("金额差异", "HIGH"),
        STATUS_DIFF("状态差异", "MEDIUM"),
        TIME_DIFF("时间差异", "LOW"),
        MISSING_INTERNAL("内部缺失", "HIGH"),
        MISSING_EXTERNAL("外部缺失", "HIGH"),
        DUPLICATE_INTERNAL("内部重复", "MEDIUM"),
        DUPLICATE_EXTERNAL("外部重复", "MEDIUM"),
        DUPLICATE_BOTH("双方重复", "HIGH"),
        FEE_DIFF("手续费差异", "MEDIUM"),
        CURRENCY_DIFF("币种差异", "HIGH");
        
        private final String description;
        private final String defaultLevel;
        
        DiffType(String description, String defaultLevel) {
            this.description = description;
            this.defaultLevel = defaultLevel;
        }
    }
    
    /**
     * 检测差异
     */
    public List<DiffRecord> detect(MatchResult matchResult, DiffRule rule) {
        List<DiffRecord> diffs = new ArrayList<>();
        
        switch (matchResult.getType()) {
            case EXACT_MATCHED:
                // 精确匹配成功，无差异
                break;
                
            case AMOUNT_DIFF:
                diffs.add(createAmountDiff(matchResult, rule));
                break;
                
            case MISSING_INTERNAL:
                diffs.add(createMissingDiff(matchResult, DiffType.MISSING_INTERNAL));
                break;
                
            case MISSING_EXTERNAL:
                diffs.add(createMissingDiff(matchResult, DiffType.MISSING_EXTERNAL));
                break;
                
            case FUZZY_MATCHED:
                // 检查是否有其他差异
                if (hasStatusDiff(matchResult)) {
                    diffs.add(createStatusDiff(matchResult));
                }
                if (hasTimeDiff(matchResult, rule.getTimeTolerance())) {
                    diffs.add(createTimeDiff(matchResult));
                }
                break;
                
            default:
                // 未知匹配类型，记录异常
                diffs.add(createUnknownDiff(matchResult));
        }
        
        return diffs;
    }
    
    /**
     * 创建金额差异记录
     */
    private DiffRecord createAmountDiff(MatchResult matchResult, DiffRule rule) {
        DiffRecord record = new DiffRecord();
        record.setDiffType(DiffType.AMOUNT_DIFF);
        record.setInternalId(matchResult.getInternal().getId());
        record.setExternalId(matchResult.getExternal().getId());
        record.setInternalAmount(matchResult.getInternal().getAmount());
        record.setExternalAmount(matchResult.getExternal().getAmount());
        record.setAmountDiff(matchResult.getInternal().getAmount()
            .subtract(matchResult.getExternal().getAmount()).abs());
        record.setTransactionId(matchResult.getInternal().getTransactionId());
        
        // 根据差异金额确定等级
        BigDecimal diffAmount = record.getAmountDiff();
        if (diffAmount.compareTo(rule.getHighThreshold()) > 0) {
            record.setDiffLevel("HIGH");
        } else if (diffAmount.compareTo(rule.getMediumThreshold()) > 0) {
            record.setDiffLevel("MEDIUM");
        } else {
            record.setDiffLevel("LOW");
        }
        
        // 分析差异原因
        record.setDiffReason(analyzeAmountDiffReason(matchResult));
        
        return record;
    }
    
    /**
     * 分析金额差异原因
     */
    private String analyzeAmountDiffReason(MatchResult matchResult) {
        BigDecimal internalAmount = matchResult.getInternal().getAmount();
        BigDecimal externalAmount = matchResult.getExternal().getAmount();
        BigDecimal diff = internalAmount.subtract(externalAmount).abs();
        
        // 常见差异原因分析
        if (diff.compareTo(new BigDecimal("0.01")) <= 0) {
            return "分分差（精度误差）";
        } else if (diff.compareTo(new BigDecimal("1")) <= 0) {
            return "手续费差异";
        } else if (internalAmount.compareTo(externalAmount) > 0) {
            return "内部金额大于外部（可能包含手续费）";
        } else {
            return "外部金额大于内部（可能包含优惠）";
        }
    }
    
    /**
     * 差异分级规则
     */
    public String classifyDiffLevel(DiffRecord diff) {
        switch (diff.getDiffType()) {
            case AMOUNT_DIFF:
                return classifyAmountDiffLevel(diff);
            case MISSING_INTERNAL:
            case MISSING_EXTERNAL:
                return "HIGH";
            case STATUS_DIFF:
                return "MEDIUM";
            case TIME_DIFF:
                return "LOW";
            default:
                return "MEDIUM";
        }
    }
    
    private String classifyAmountDiffLevel(DiffRecord diff) {
        BigDecimal amount = diff.getAmountDiff();
        
        if (amount.compareTo(new BigDecimal("10000")) > 0) {
            return "CRITICAL";  // 严重
        } else if (amount.compareTo(new BigDecimal("1000")) > 0) {
            return "HIGH";      // 高
        } else if (amount.compareTo(new BigDecimal("100")) > 0) {
            return "MEDIUM";    // 中
        } else {
            return "LOW";       // 低
        }
    }
}
```

```java
/**
 * 差异自动处理引擎
 */
@Component
public class DiffAutoHandler {
    
    @Autowired
    private ReconciliationRuleEngine ruleEngine;
    
    @Autowired
    private AccountingService accountingService;
    
    /**
     * 自动处理差异
     */
    @Transactional
    public HandleResult autoHandle(DiffRecord diff) {
        // 获取处理策略
        HandleStrategy strategy = ruleEngine.getHandleStrategy(
            diff.getTask().getTaskType(), 
            diff.getDiffType().name()
        );
        
        if (!strategy.isAutoHandle()) {
            return HandleResult.skip("该类型差异需要人工处理");
        }
        
        switch (diff.getDiffType()) {
            case AMOUNT_DIFF:
                return handleAmountDiff(diff, strategy);
            case MISSING_INTERNAL:
                return handleMissingInternal(diff, strategy);
            case MISSING_EXTERNAL:
                return handleMissingExternal(diff, strategy);
            case STATUS_DIFF:
                return handleStatusDiff(diff, strategy);
            case DUPLICATE_INTERNAL:
            case DUPLICATE_EXTERNAL:
            case DUPLICATE_BOTH:
                return handleDuplicate(diff, strategy);
            default:
                return HandleResult.skip("未知差异类型");
        }
    }
    
    /**
     * 处理金额差异
     */
    private HandleResult handleAmountDiff(DiffRecord diff, HandleStrategy strategy) {
        BigDecimal threshold = strategy.getAutoHandleThreshold();
        
        // 差异金额在阈值内，自动调账
        if (diff.getAmountDiff().compareTo(threshold) <= 0) {
            // 创建调账凭证
            AdjustmentVoucher voucher = createAdjustmentVoucher(diff);
            
            // 提交账务处理
            accountingService.postVoucher(voucher);
            
            // 更新差异状态
            diff.setHandleStatus("PROCESSED");
            diff.setAccountingVoucherNo(voucher.getVoucherNo());
            
            return HandleResult.success("已自动调账，凭证号：" + voucher.getVoucherNo());
        } else {
            return HandleResult.skip("差异金额超出自动处理阈值");
        }
    }
    
    /**
     * 创建调账凭证
     */
    private AdjustmentVoucher createAdjustmentVoucher(DiffRecord diff) {
        AdjustmentVoucher voucher = new AdjustmentVoucher();
        voucher.setVoucherNo(generateVoucherNo());
        voucher.setDiffId(diff.getId());
        voucher.setTaskId(diff.getTaskId());
        voucher.setAdjustmentType("SUPPLEMENT");
        voucher.setAccountNo(diff.getInternal().getAccountNo());
        
        // 根据差异方向确定借贷
        if (diff.getInternalAmount().compareTo(diff.getExternalAmount()) > 0) {
            // 内部大于外部，需要冲销
            voucher.setDebitAmount(BigDecimal.ZERO);
            voucher.setCreditAmount(diff.getAmountDiff());
            voucher.setSummary("对账差异冲销：" + diff.getTransactionId());
        } else {
            // 内部小于外部，需要补记
            voucher.setDebitAmount(diff.getAmountDiff());
            voucher.setCreditAmount(BigDecimal.ZERO);
            voucher.setSummary("对账差异补记：" + diff.getTransactionId());
        }
        
        voucher.setCreator("SYSTEM");
        voucher.setStatus("PENDING");
        
        return voucher;
    }
    
    /**
     * 处理内部缺失
     */
    private HandleResult handleMissingInternal(DiffRecord diff, HandleStrategy strategy) {
        if (strategy.isAutoSupplement()) {
            // 自动补录内部记录
            ReconciliationData supplementData = buildSupplementData(diff.getExternal());
            
            // 保存补录数据
            reconciliationDataRepository.save(supplementData);
            
            // 重新匹配
            return HandleResult.success("已自动补录内部记录");
        } else {
            return HandleResult.skip("内部缺失需要人工确认");
        }
    }
    
    /**
     * 处理重复数据
     */
    private HandleResult handleDuplicate(DiffRecord diff, HandleStrategy strategy) {
        if (strategy.isAutoDedup()) {
            // 根据规则去重，保留最新的记录
            List<ReconciliationData> duplicates = findDuplicates(diff);
            
            if (duplicates.size() > 1) {
                // 保留第一条，标记其他为重复
                for (int i = 1; i < duplicates.size(); i++) {
                    duplicates.get(i).setStatus("DUPLICATE");
                    reconciliationDataRepository.save(duplicates.get(i));
                }
                
                return HandleResult.success("已自动去重，保留最新记录");
            }
        }
        
        return HandleResult.skip("重复数据需要人工确认");
    }
}
```

```java
/**

@Service
@Slf4j
public class BankReconciliationService {
    
    @Autowired
    private BankFileParser fileParser;
    
    @Autowired
    private MultiDimensionMatcher matcher;
    
    @Autowired
    private DiffDetectionEngine diffEngine;
    
    @Autowired
    private DiffAutoHandler autoHandler;
    
    /**
     * 执行银行对账
     */
    @Transactional
    public ReconciliationReport reconcile(LocalDate reconciliationDate) {
        log.info("开始银行对账，对账日期：{}", reconciliationDate);
        
        // 1. 创建对账任务
        ReconciliationTask task = createTask(reconciliationDate);
        
        try {
            // 2. 获取内部数据
            List<ReconciliationData> internalData = getInternalData(reconciliationDate);
            log.info("获取内部数据{}条", internalData.size());
            
            // 3. 获取外部数据（银行对账文件）
            List<ReconciliationData> externalData = getExternalData(reconciliationDate);
            log.info("获取外部数据{}条", externalData.size());
            
            // 4. 数据预处理
            internalData = preprocessData(internalData);
            externalData = preprocessData(externalData);
            
            // 5. 执行匹配
            List<MatchResult> matchResults = matcher.exactMatch(internalData, externalData);
            log.info("匹配完成，总计{}条，匹配成功{}条", 
                matchResults.size(),
                matchResults.stream().filter(m -> m.getType() == MatchType.EXACT_MATCHED).count());
            
            // 6. 检测差异
            List<DiffRecord> diffs = new ArrayList<>();
            for (MatchResult matchResult : matchResults) {
                diffs.addAll(diffEngine.detect(matchResult, getDiffRule()));
            }
            log.info("检测到差异{}条", diffs.size());
            
            // 7. 自动处理差异
            for (DiffRecord diff : diffs) {
                HandleResult result = autoHandler.autoHandle(diff);
                log.info("差异处理结果：{}", result);
            }
            
            // 8. 保存匹配和差异数据
            saveMatchResults(matchResults);
            saveDiffRecords(diffs);
            
            // 9. 生成对账报告
            ReconciliationReport report = generateReport(task, matchResults, diffs);
            
            // 10. 更新任务状态
            task.setStatus("SUCCESS");
            task.setEndTime(LocalDateTime.now());
            task.setMatchedCount(matchResults.stream()
                .filter(m -> m.getType() == MatchType.EXACT_MATCHED).count());
            task.setDiffCount(diffs.size());
            taskRepository.save(task);
            
            log.info("银行对账完成，任务号：{}", task.getTaskNo());
            return report;
            
        } catch (Exception e) {
            log.error("银行对账失败", e);
            task.setStatus("FAILED");
            taskRepository.save(task);
            throw new ReconciliationException("银行对账失败", e);
        }
    }
    
    /**
     * 解析银行对账文件
     */
    private List<ReconciliationData> parseBankFile(String filePath, String bankCode) {
        BankFileParser parser = BankFileParserFactory.getParser(bankCode);
        
        // 读取文件
        List<String> lines = FileUtil.readLines(filePath);
        
        // 解析数据
        return lines.stream()
            .filter(line -> !line.startsWith("HEADER") && !line.startsWith("TRAILER"))
            .map(line -> parser.parse(line))
            .filter(Objects::nonNull)
            .collect(Collectors.toList());
    }
    
    /**
     * 银行对账差异分析
     */
    private String analyzeBankDiff(DiffRecord diff) {
        if (diff.getDiffType() == DiffType.MISSING_EXTERNAL) {
            // 银行无我有
            return analyzeMissingBank(diff);
        } else if (diff.getDiffType() == DiffType.MISSING_INTERNAL) {
            // 银行有我无
            return analyzeMissingInternal(diff);
        } else if (diff.getDiffType() == DiffType.AMOUNT_DIFF) {
            // 金额不一致
            return analyzeAmountDiff(diff);
        }
        return "未知差异";
    }
    
    private String analyzeMissingBank(DiffRecord diff) {
        // 分析银行无记录的原因
        ReconciliationData internal = diff.getInternal();
        
        if ("PENDING".equals(internal.getStatus())) {
            return "交易处理中，银行尚未完成清算";
        } else if ("FAILED".equals(internal.getStatus())) {
            return "交易失败，银行已拒绝";
        } else if ("REFUNDED".equals(internal.getStatus())) {
            return "交易已退款，银行已处理";
        }
        return "需要与银行确认";
    }
    
    private String analyzeMissingInternal(DiffRecord diff) {
        // 分析内部无记录的原因
        ReconciliationData external = diff.getExternal();
        
        // 检查是否是银行手续费
        if (external.getTransactionType().contains("FEE")) {
            return "银行手续费，内部未记录";
        }
        
        // 检查是否是利息
        if (external.getTransactionType().contains("INTEREST")) {
            return "银行利息，内部未记录";
        }
        
        return "需要核实交易来源";
    }
```

```sql
-- =====================================================
-- 银行对账综合查询
-- =====================================================
SELECT 
    a.transaction_id,
    a.transaction_time as internal_time,
    b.transaction_time as external_time,
    a.amount as internal_amount,
    b.amount as external_amount,
    a.status as internal_status,
    b.status as external_status,
    CASE 
        WHEN b.transaction_id IS NULL THEN '银行无记录'
        WHEN a.transaction_id IS NULL THEN '内部无记录'
        WHEN a.amount = b.amount AND a.status = b.status THEN '完全匹配'
        WHEN a.amount = b.amount THEN '金额匹配，状态不同'
        WHEN a.status = b.status THEN '状态匹配，金额不同'
        ELSE '金额和状态均不同'
    END as match_result,
    CASE 
        WHEN b.transaction_id IS NULL THEN 'MISSING_EXTERNAL'
        WHEN a.transaction_id IS NULL THEN 'MISSING_INTERNAL'
        WHEN a.amount != b.amount THEN 'AMOUNT_DIFF'
        WHEN a.status != b.status THEN 'STATUS_DIFF'
        ELSE 'MATCHED'
    END as diff_type
FROM internal_transaction a
LEFT JOIN external_transaction b 
    ON a.transaction_id = b.transaction_id
WHERE a.transaction_date = '2026-09-10'
   OR b.transaction_date = '2026-09-10'
ORDER BY match_result, a.transaction_time;

-- =====================================================
-- 对账差异统计
-- =====================================================
SELECT 
    diff_type,
    COUNT(*) as diff_count,
    SUM(amount_diff) as total_amount,
    AVG(amount_diff) as avg_amount,
    MAX(amount_diff) as max_amount
FROM reconciliation_diff
WHERE task_id = #{taskId}
GROUP BY diff_type
ORDER BY total_amount DESC;

-- =====================================================
-- 对账进度监控
-- =====================================================
SELECT 
    DATE_FORMAT(transaction_date, '%Y-%m') as month,
    COUNT(*) as total_tasks,
    SUM(CASE WHEN status = 'SUCCESS' THEN 1 ELSE 0 END) as success_count,
    SUM(CASE WHEN status = 'FAILED' THEN 1 ELSE 0 END) as failed_count,
    SUM(CASE WHEN status = 'PARTIAL' THEN 1 ELSE 0 END) as partial_count,
    ROUND(SUM(CASE WHEN status = 'SUCCESS' THEN 1 ELSE 0 END) / COUNT(*) * 100, 2) as success_rate
FROM reconciliation_task
WHERE task_type = 'BANK'
GROUP BY DATE_FORMAT(transaction_date, '%Y-%m')
ORDER BY month DESC;
```

### 7.2 分布式对账

```java
/**
 * 分布式对账调度
 */
@Component
public class DistributedReconciliationScheduler {
    
    @Autowired
    private TaskScheduler taskScheduler;
    
    @Autowired
    private ShardingService shardingService;
    
    /**
     * 分布式对账任务调度
     */
    public void scheduleReconciliation(LocalDate date, String taskType) {
        // 获取分片信息
        List<ShardInfo> shards = shardingService.getShards("reconciliation_" + taskType);
        
        // 并行执行分片任务
        List<Future<ShardResult>> futures = new ArrayList<>();
        
        for (ShardInfo shard : shards) {
            futures.add(taskScheduler.submit(() -> executeShard(date, taskType, shard)));
        }
        
        // 等待所有分片完成
        List<ShardResult> results = new ArrayList<>();
        for (Future<ShardResult> future : futures) {
            try {
                results.add(future.get());
            } catch (Exception e) {
                log.error("分片任务执行失败", e);
            }
        }
        
        // 合并结果
        mergeShardResults(results);
    }
    
    /**
     * 执行分片任务
     */
    private ShardResult executeShard(LocalDate date, String taskType, ShardInfo shard) {
        // 根据分片获取数据范围
        DataRange range = shard.getDataRange();
        
        // 获取分片数据
        List<ReconciliationData> internalData = getInternalData(date, range);
        List<ReconciliationData> externalData = getExternalData(date, range);
        
        // 执行匹配
        List<MatchResult> matchResults = matcher.exactMatch(internalData, externalData);
        
        // 检测差异
        List<DiffRecord> diffs = detectDiffs(matchResults);
        
        // 保存结果
        saveResults(matchResults, diffs, shard);
        
        return ShardResult.success(shard.getId(), matchResults.size(), diffs.size());
    }
}
```
